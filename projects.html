10. Reverse of a Number

#include <stdio.h>

int main() {
    int n, rev = 0, rem;

    scanf("%d", &n);

    while(n != 0) {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    printf("Reverse = %d", rev);
    return 0;
}
Input: 1234
Output: Reverse = 4321

11. Fibonacci Series

#include <stdio.h>

int main() {
    int n, i;
    int a = 0, b = 1, next;

    scanf("%d", &n);

    printf("Fibonacci Series: ");
    for(i = 1; i <= n; i++) {
        printf("%d ", a);
        next = a + b;
        a = b;
        b = next;
    }

    return 0;
}
Input: 5
Output: Fibonacci Series: 0 1 1 2 3

12. GCD & LCM of Two Numbers

#include <stdio.h>

int main() {
    int a, b, x, y, gcd, lcm;

    scanf("%d %d", &a, &b);

    x = a;
    y = b;

    while(b != 0) {
        int temp = b;
        b = a % b;
        a = temp;
    }
    gcd = a;

    lcm = (x * y) / gcd;

    printf("GCD = %d\n", gcd);
    printf("LCM = %d", lcm);

    return 0;
}
Input: 12 18
Output: GCD = 6 LCM = 36

13. Student Structure (Array + Functions)

#include <stdio.h>

struct Student {
    int roll_no;
    char name[50];
    float marks;
};

void input(struct Student s[], int n) {
    for(int i = 0; i < n; i++) {
        printf("Enter details of student %d:\n", i + 1);

        printf("Roll No: ");
        scanf("%d", &s[i].roll_no);

        printf("Name: ");
        scanf("%s", s[i].name);

        printf("Marks: ");
        scanf("%f", &s[i].marks);
    }
}

void display(struct Student s[], int n) {
    printf("\nStudent Details:\n");

    for(int i = 0; i < n; i++) {
        printf("\nStudent %d\n", i + 1);
        printf("Roll No: %d\n", s[i].roll_no);
        printf("Name: %s\n", s[i].name);
        printf("Marks: %.2f\n", s[i].marks);
    }
}

int main() {
    struct Student s[3];

    input(s, 3);
    display(s, 3);

    return 0;
}
Input: 1 Ankit 85 2 Vikend 90 3 Aditya 78
Output: Student Details Student 1 → Ankit (85) Student 2 → Vikend (90) Student 3 → Aditya (78)
